# PCRE2 configuration for OpenVMS (VSI C, IA64 and x86-64), applied to
# src/config.h.generic by tools/gen_config.py.  One setting per line:
#   define NAME [VALUE]     (VALUE defaults to 1)
#   undef NAME
# Settings not listed keep the generic file's value.  Header/function
# answers follow the CRTL probes in vms-grep (docs/vms-environment.md there).

# What to build: the 8-bit library with Unicode support, as a static library.
define SUPPORT_PCRE2_8
define SUPPORT_UNICODE
define PCRE2_STATIC
# No JIT: SLJIT has no OpenVMS support.
undef SUPPORT_JIT

# Headers present in the VSI C RTL.
define HAVE_ASSERT_H
define HAVE_DIRENT_H
define HAVE_INTTYPES_H
define HAVE_LIMITS_H
define HAVE_STDINT_H
define HAVE_STDIO_H
define HAVE_STDLIB_H
define HAVE_STRINGS_H
define HAVE_STRING_H
define HAVE_SYS_STAT_H
define HAVE_SYS_TYPES_H
define HAVE_SYS_WAIT_H
define HAVE_UNISTD_H
define HAVE_WCHAR_H
define STDC_HEADERS

# Functions present in the VSI C RTL (setrlimit, mkostemp and secure_getenv are not).
define HAVE_REALPATH

# VSI C 7.x has no __builtin_* or __attribute__ support: leave
# HAVE_BUILTIN_*, HAVE_ATTRIBUTE_UNINITIALIZED and HAVE_VISIBILITY undefined.

# VSI C on IA64 warns that printf's %zu/%td are "not supported or not fully
# supported" (they work at run time); pcre2test then uses %lu/%ld instead.
define DISABLE_PERCENT_ZT
